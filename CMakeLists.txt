cmake_minimum_required(VERSION 3.20)
project(PSTouchNativeCorePOC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
add_library(pstouch_image_core src/image.cpp src/image_io.cpp src/document.cpp src/psd.cpp)
target_link_libraries(pstouch_image_core PUBLIC PNG::PNG JPEG::JPEG)
target_include_directories(pstouch_image_core PUBLIC include)
if(MSVC)
  target_compile_options(pstouch_image_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(pstouch_image_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(pstouch-image src/main.cpp)
target_link_libraries(pstouch-image PRIVATE pstouch_image_core)
add_executable(pstouch_core_tests tests/test_image.cpp)
target_link_libraries(pstouch_core_tests PRIVATE pstouch_image_core)
add_executable(pstouch_image_io_tests tests/test_image_io.cpp)
target_link_libraries(pstouch_image_io_tests PRIVATE pstouch_image_core)
add_executable(pstouch_document_tests tests/test_document.cpp)
target_link_libraries(pstouch_document_tests PRIVATE pstouch_image_core)
add_executable(pstouch_psd_tests tests/test_psd.cpp)
target_link_libraries(pstouch_psd_tests PRIVATE pstouch_image_core)
enable_testing()
add_test(NAME pstouch_core_tests COMMAND pstouch_core_tests)
add_test(NAME pstouch_image_io_tests COMMAND pstouch_image_io_tests)
add_test(NAME pstouch_document_tests COMMAND pstouch_document_tests)
add_test(NAME pstouch_psd_tests COMMAND pstouch_psd_tests)
add_executable(pstouch_psd_import_tests tests/test_psd_import.cpp)
target_link_libraries(pstouch_psd_import_tests PRIVATE pstouch_image_core)
add_test(NAME pstouch_psd_import_tests COMMAND pstouch_psd_import_tests)
add_executable(pstouch_document_ops_tests tests/test_document_ops.cpp)
target_link_libraries(pstouch_document_ops_tests PRIVATE pstouch_image_core)
add_test(NAME pstouch_document_ops_tests COMMAND pstouch_document_ops_tests)

add_executable(pstouch_image_ops_tests tests/test_image_ops.cpp)
target_link_libraries(pstouch_image_ops_tests PRIVATE pstouch_image_core)
add_test(NAME pstouch_image_ops_tests COMMAND pstouch_image_ops_tests)

if(WIN32)
  add_executable(PS-Touch WIN32 windows/pstouch_win32.cpp)
  target_compile_features(PS-Touch PRIVATE cxx_std_17)
  target_link_libraries(PS-Touch PRIVATE pstouch_image_core gdiplus comdlg32)
  if(MSVC)
    target_compile_options(PS-Touch PRIVATE /W4 /permissive- /utf-8)
  endif()
  set_target_properties(PS-Touch PROPERTIES OUTPUT_NAME "PS-Touch")
endif()
